cmake_minimum_required(VERSION 3.22)
project(rlhd_shader LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_PCH OFF CACHE BOOL "" FORCE)
set(ENABLE_RTTI ON CACHE BOOL "" FORCE)
set(ENABLE_EXCEPTIONS ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_MSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_UTIL OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/glslang)
add_subdirectory(vendor/spirv-cross)
add_library(rlhd_shader SHARED shader_translate.cpp)
target_include_directories(rlhd_shader PRIVATE "$ENV{JAVA_HOME}/include" "$ENV{JAVA_HOME}/include/linux" vendor/glslang vendor/spirv-cross)
target_compile_options(rlhd_shader PRIVATE -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_options(rlhd_shader PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,--gc-sections")
target_link_libraries(rlhd_shader PRIVATE glslang glslang-default-resource-limits spirv-cross-glsl)
if(ANDROID)
 target_link_options(rlhd_shader PRIVATE "-Wl,-z,max-page-size=16384")
endif()
